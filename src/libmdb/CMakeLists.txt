cmake_minimum_required(VERSION 3.25)

set ( TARGET libmdb )
if ( ${CMAKE_SYSTEM_NAME} STREQUAL "Windows" )
    set ( TARGET_NAME ${TARGET} )
else ()
    set ( TARGET_NAME mdb )
endif ()
set( PRODUCT_NAME "mdbtools" )
set( PRODUCT_COMPANY_NAME "" )
set( PRODUCT_COPYRIGHT "Copyright \\251 2000-2011 Brian Bruns and others" )
set( PRODUCT_VER_MAJ ${PROJECT_VERSION_MAJOR} )
set( PRODUCT_VER_MIN ${PROJECT_VERSION_MINOR} )
set( PRODUCT_VER_PAT ${PROJECT_VERSION_PATCH} )
set( PRODUCT_VER_REV ${REVISION_NUMBER} )
message( "Version for product ${PRODUCT_NAME} was set to ${PRODUCT_VER_MAJ}.${PRODUCT_VER_MIN}.${PRODUCT_VER_PAT}.${PRODUCT_VER_REV}")
set( TARGET_VER_MAJ ${PROJECT_VERSION_MAJOR} )
set( TARGET_VER_MIN ${PROJECT_VERSION_MINOR} )
set( TARGET_VER_PAT ${PROJECT_VERSION_PATCH} )
set( TARGET_VER_REV ${REVISION_NUMBER} )
set( TARGET_DESCRIPTION "LibMDB - MDB Tools" )
message( "Version for target ${TARGET_NAME} was set to ${TARGET_VER_MAJ}.${TARGET_VER_MIN}.${TARGET_VER_PAT}.${TARGET_VER_REV}")

# Configure header files from .in templates
include(CheckIncludeFile)
include(CheckTypeSize)
include(CheckFunctionExists)

# Set feature test macros for POSIX functions (needed for wcstombs_l, newlocale, etc.)
if(APPLE)
    # macOS requires _DARWIN_C_SOURCE and _USE_EXTENDED_LOCALES_
    set(CMAKE_REQUIRED_DEFINITIONS "-D_DARWIN_C_SOURCE -D_USE_EXTENDED_LOCALES_")
elseif(NOT WIN32)
    set(CMAKE_REQUIRED_DEFINITIONS "-D_XOPEN_SOURCE=700 -D_POSIX_C_SOURCE=200809L")
endif()

# Check for required headers
if(APPLE)
    # On macOS, use xlocale instead of iconv
    set(HAVE_ICONV_H OFF)
    set(HAVE_XLOCALE_H ON)
else()
    check_include_file("iconv.h" HAVE_ICONV_H)
    check_include_file("xlocale.h" HAVE_XLOCALE_H)
endif()
check_include_file("glib.h" HAVE_GLIB_H)
check_function_exists("strptime" HAVE_STRPTIME)
check_function_exists("fmemopen" HAVE_FMEMOPEN)
check_function_exists("gmtime_r" HAVE_GMTIME_R)
check_function_exists("wcstombs_l" HAVE_WCSTOMBS_L)
check_function_exists("mbstowcs_l" HAVE_MBSTOWCS_L)
check_function_exists("reallocf" HAVE_REALLOCF)
check_function_exists("vasprintf" HAVE_VASPRINTF)
check_function_exists("vasnprintf" HAVE_VASNPRINTF)

# Set configuration variables for mdbtools.h
if(HAVE_ICONV_H)
    set(HAVE_ICONV_H 1)
else()
    set(HAVE_ICONV_H 0)
endif()

if(HAVE_XLOCALE_H)
    set(HAVE_XLOCALE_H 1)
else()
    set(HAVE_XLOCALE_H 0)
endif()

# Set GLib header
if(HAVE_GLIB_H)
    set(GLIB_INCLUDE_HEADER "glib.h")
else()
    set(GLIB_INCLUDE_HEADER "mdbfakeglib.h")
endif()

# Detect locale_t type
check_type_size("locale_t" LOCALE_T_SIZE)
if(HAVE_LOCALE_T_SIZE)
    set(LOCALE_T "locale_t")
else()
    # Fallback for systems without locale_t
    set(LOCALE_T "void*")
endif()

# Create build include directory
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/include")

# Configure header files
configure_file(
    "${MDBTOOLS_INCLUDE_DIR}/mdbtools.h.in"
    "${CMAKE_CURRENT_BINARY_DIR}/include/mdbtools.h"
    @ONLY
)

# Set version variables for mdbver.h
set(PACKAGE "${PRODUCT_NAME}")
set(VERSION "${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR}.${PROJECT_VERSION_PATCH}")

configure_file(
    "${MDBTOOLS_INCLUDE_DIR}/mdbver.h.in"
    "${CMAKE_CURRENT_BINARY_DIR}/include/mdbver.h"
    @ONLY
)

# Create header wrappers for Windows (strings.h, unistd.h, getopt.h don't exist on Windows)
if(${CMAKE_SYSTEM_NAME} STREQUAL "Windows")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/include/strings.h"
"/* strings.h wrapper for Windows - generated by CMake */
#ifndef _STRINGS_H_WRAPPER_
#define _STRINGS_H_WRAPPER_
/* MSVC/Windows compatibility */
#include <string.h>
#include <intrin.h>
#define strcasecmp _stricmp
#define strtok_r strtok_s
#define fseeko(f, o, w) fseek(f, (long)(o), w)
#define ftello(f) ftell(f)
#define __builtin_bswap32(x) _byteswap_ulong(x)
#include <stdlib.h>
#endif /* _STRINGS_H_WRAPPER_ */
")
    message(STATUS "Generated strings.h wrapper for Windows")

    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/include/unistd.h"
"/* unistd.h wrapper for Windows - generated by CMake */
#ifndef _UNISTD_H_WRAPPER_
#define _UNISTD_H_WRAPPER_

#include <windows.h>
#define access _access
#define F_OK 0
#define R_OK 4
#define W_OK 2
#define X_OK 1
#define unlink _unlink
#define read _read
#define write _write
#define close _close
#include <BaseTsd.h>
typedef SSIZE_T ssize_t;

#endif /* _UNISTD_H_WRAPPER_ */
")
    message(STATUS "Generated unistd.h wrapper for Windows")

    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/include/getopt.h"
"/* getopt.h wrapper for Windows - generated by CMake */
#ifndef _GETOPT_H_WRAPPER_
#define _GETOPT_H_WRAPPER_
#include <ya_getopt.h>
#endif /* _GETOPT_H_WRAPPER_ */
")
    message(STATUS "Generated getopt.h wrapper for Windows")
endif()

message(STATUS "Generated mdbtools.h and mdbver.h from templates")

# Collect libmdb source files from mdbtools
set ( SOURCES )  # Ensure the variable is empty
list(APPEND SOURCES
    ${CMAKE_CURRENT_LIST_DIR}/backend.c
    ${CMAKE_CURRENT_LIST_DIR}/catalog.c
    ${CMAKE_CURRENT_LIST_DIR}/data.c
    ${CMAKE_CURRENT_LIST_DIR}/dump.c
    ${CMAKE_CURRENT_LIST_DIR}/file.c
    ${CMAKE_CURRENT_LIST_DIR}/iconv.c
    ${CMAKE_CURRENT_LIST_DIR}/index.c
    ${CMAKE_CURRENT_LIST_DIR}/like.c
    ${CMAKE_CURRENT_LIST_DIR}/map.c
    ${CMAKE_CURRENT_LIST_DIR}/money.c
    ${CMAKE_CURRENT_LIST_DIR}/options.c
    ${CMAKE_CURRENT_LIST_DIR}/props.c
    ${CMAKE_CURRENT_LIST_DIR}/rc4.c
    ${CMAKE_CURRENT_LIST_DIR}/sargs.c
    ${CMAKE_CURRENT_LIST_DIR}/stats.c
    ${CMAKE_CURRENT_LIST_DIR}/table.c
    ${CMAKE_CURRENT_LIST_DIR}/version.c
    ${CMAKE_CURRENT_LIST_DIR}/worktable.c
    ${CMAKE_CURRENT_LIST_DIR}/write.c
)
if(NOT HAVE_GLIB_H)
    list(APPEND SOURCES
        ${CMAKE_CURRENT_LIST_DIR}/fakeglib.c
    )
endif()
if(${CMAKE_SYSTEM_NAME} STREQUAL "Windows")
    # Add Windows-specific source files if needed)
    list(APPEND SOURCES
        ${MESTRELAB_DIR}/ya_getopt/ya_getopt.c
    )
endif()

if ( ${CMAKE_SYSTEM_NAME} STREQUAL "Windows" )
    configure_file(
        ${MESTRELAB_DIR}/libmdb.rc.in
        ${CMAKE_CURRENT_BINARY_DIR}/${TARGET}.rc
    @ONLY)
    list(APPEND SOURCES
        ${MESTRELAB_DIR}/resource.h
        ${CMAKE_CURRENT_BINARY_DIR}/${TARGET}.rc
    )
elseif( ${CMAKE_SYSTEM_NAME} STREQUAL "Linux" )
    # in linux some info is added to the .comment section of the ELF binary
    # use this command to print this information
    # readelf -p .comment /path/to/SA*.so
    configure_file(
        ${MESTRELAB_DIR}/version_linux.c.in
        ${CMAKE_CURRENT_BINARY_DIR}/${TARGET}_version.c
    @ONLY)
    list(APPEND SOURCES
        ${CMAKE_CURRENT_BINARY_DIR}/${TARGET}_version.c
    )
endif()

add_library ( ${TARGET_NAME} SHARED ${SOURCES} )

# On Windows, automatically export all symbols (generates .lib)
if(MSVC)
    set_target_properties(${TARGET_NAME} PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
    # Disable module definition file generation to use WINDOWS_EXPORT_ALL_SYMBOLS instead
    #set_target_properties(${TARGET_NAME} PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS_REPLACES_EXPORTS ON)
endif()

# Configure include directories
target_include_directories(
    ${TARGET_NAME}
    PUBLIC
    "${CMAKE_CURRENT_BINARY_DIR}/include"  # Generated headers first
    $<$<PLATFORM_ID:Windows>:${MESTRELAB_DIR}/ya_getopt>
)

target_include_directories(${TARGET_NAME} PUBLIC ${MDBTOOLS_INCLUDE_DIR})

# Add feature test macros for POSIX functions
if(APPLE)
    # macOS requires _DARWIN_C_SOURCE and _USE_EXTENDED_LOCALES_ to expose locale functions
    target_compile_definitions(${TARGET_NAME} PRIVATE
        _DARWIN_C_SOURCE
        _USE_EXTENDED_LOCALES_
    )
elseif(NOT WIN32)
    target_compile_definitions(${TARGET_NAME} PRIVATE
        _XOPEN_SOURCE=700
        _POSIX_C_SOURCE=200809L
    )
endif()

if (HAVE_STRPTIME)
    target_compile_definitions(${TARGET_NAME} PRIVATE HAVE_STRPTIME)
endif()
if(HAVE_FMEMOPEN)
    target_compile_definitions(${TARGET_NAME} PRIVATE HAVE_FMEMOPEN)
endif()
if (HAVE_GMTIME_R)
    target_compile_definitions(${TARGET_NAME} PRIVATE HAVE_GMTIME_R)
endif()
if (HAVE_WCSTOMBS_L)
    target_compile_definitions(${TARGET_NAME} PRIVATE HAVE_WCSTOMBS_L)
endif()
if (HAVE_MBSTOWCS_L)
    target_compile_definitions(${TARGET_NAME} PRIVATE HAVE_MBSTOWCS_L)
endif()
if (HAVE_REALLOCF)
    target_compile_definitions(${TARGET_NAME} PRIVATE HAVE_REALLOCF)
endif()
if (HAVE_VASPRINTF)
    target_compile_definitions(${TARGET_NAME} PRIVATE HAVE_VASPRINTF)
endif()
if (HAVE_VASNPRINTF)
    target_compile_definitions(${TARGET_NAME} PRIVATE HAVE_VASNPRINTF)
endif()

if ( ${CMAKE_SYSTEM_NAME} STREQUAL "Windows" )
    if(DEFINED OUTPUT_NAME_SUFFIX)
        set_target_properties(${TARGET_NAME} PROPERTIES OUTPUT_NAME "${TARGET}${OUTPUT_NAME_SUFFIX}")
    endif()
else ()
    if(DEFINED OUTPUT_NAME_SUFFIX)
        set_target_properties(${TARGET_NAME} PROPERTIES OUTPUT_NAME "mdb${OUTPUT_NAME_SUFFIX}")
    endif()
endif ()

set_property(TARGET ${TARGET_NAME} PROPERTY C_STANDARD 11)

# Define TLS support based on compiler
if(MSVC)
    target_compile_definitions(${TARGET_NAME} PRIVATE "TLS=__declspec(thread)")
elseif(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_definitions(${TARGET_NAME} PRIVATE "TLS=__thread")
else()
    target_compile_definitions(${TARGET_NAME} PRIVATE "TLS=")
endif()

if ( ${CMAKE_SYSTEM_NAME} STREQUAL "Darwin" )
    set_target_properties( ${TARGET_NAME} PROPERTIES
        MACHO_COMPATIBILITY_VERSION ${TARGET_VER_MAJ}.${TARGET_VER_MIN}.${TARGET_VER_PAT}
        MACHO_CURRENT_VERSION ${TARGET_VER_MAJ}.${TARGET_VER_MIN}.${TARGET_VER_PAT}
        SUFFIX ".so"
    )
elseif( ${CMAKE_SYSTEM_NAME} STREQUAL "Linux" )
    target_link_libraries( ${TARGET_NAME}
        PRIVATE dl
    )
endif()


set(MDBTOOLS_INSTALL_INCLUDE_DIR "${MDBTOOLS_INSTALL_DIR}/include")
if(MDB_INSTALL_ONLY_LIBRARY)
    set(MDBTOOLS_INSTALL_LIB_DIR "${MDBTOOLS_INSTALL_DIR}")
    set(MDBTOOLS_INSTALL_BIN_DIR "${MDBTOOLS_INSTALL_DIR}")
else()
    set(MDBTOOLS_INSTALL_LIB_DIR "${MDBTOOLS_INSTALL_DIR}/lib")
    set(MDBTOOLS_INSTALL_BIN_DIR "${MDBTOOLS_INSTALL_DIR}/bin")
endif()

install (
    TARGETS ${TARGET_NAME}
    LIBRARY DESTINATION ${MDBTOOLS_INSTALL_LIB_DIR}
    ARCHIVE DESTINATION ${MDBTOOLS_INSTALL_LIB_DIR}
    RUNTIME DESTINATION ${MDBTOOLS_INSTALL_BIN_DIR}
)

# Install PDB files on Windows for debugging
if(MSVC)
    install(
        FILES $<TARGET_PDB_FILE:${TARGET_NAME}>
        DESTINATION ${MDBTOOLS_INSTALL_BIN_DIR}
        OPTIONAL
    )
endif()

if (NOT MDB_INSTALL_ONLY_LIBRARY)
    # Install public headers
    install (
        FILES
            "${CMAKE_CURRENT_BINARY_DIR}/include/mdbtools.h"
            "${CMAKE_CURRENT_BINARY_DIR}/include/mdbver.h"
            "${MDBTOOLS_INCLUDE_DIR}/mdbsql.h"
        DESTINATION ${MDBTOOLS_INSTALL_INCLUDE_DIR}
    )

    # Install mdbfakeglib.h if not using system GLib
    install (
        FILES "${MDBTOOLS_INCLUDE_DIR}/mdbfakeglib.h"
        DESTINATION ${MDBTOOLS_INSTALL_INCLUDE_DIR}
    )

    # Install Windows-specific wrapper headers and ya_getopt
    if(${CMAKE_SYSTEM_NAME} STREQUAL "Windows")
        install (
            FILES
                "${CMAKE_CURRENT_BINARY_DIR}/include/strings.h"
                "${CMAKE_CURRENT_BINARY_DIR}/include/unistd.h"
                "${CMAKE_CURRENT_BINARY_DIR}/include/getopt.h"
                "${MESTRELAB_DIR}/ya_getopt/ya_getopt.h"
            DESTINATION ${MDBTOOLS_INSTALL_INCLUDE_DIR}
        )
    endif()

    # Configure and install CMake config files for find_package support
    set(CMAKE_INSTALL_PREFIX_CONFIGURED "${CMAKE_INSTALL_PREFIX}")
    set(CMAKE_INSTALL_INCLUDEDIR "include")

    configure_file(
        "${CMAKE_CURRENT_SOURCE_DIR}/mdbConfig.cmake.in"
        "${CMAKE_CURRENT_BINARY_DIR}/mdbConfig.cmake"
        @ONLY
    )

    configure_file(
        "${CMAKE_CURRENT_SOURCE_DIR}/mdbConfigVersion.cmake.in"
        "${CMAKE_CURRENT_BINARY_DIR}/mdbConfigVersion.cmake"
        @ONLY
    )

    install(
        FILES
            "${CMAKE_CURRENT_BINARY_DIR}/mdbConfig.cmake"
            "${CMAKE_CURRENT_BINARY_DIR}/mdbConfigVersion.cmake"
        DESTINATION ${MDBTOOLS_INSTALL_LIB_DIR}/cmake/mdb
    )
endif()
